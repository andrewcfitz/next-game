# Next Game for iOS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Next Game's per-team widgets on iPhone (Home Screen and Lock Screen) through TestFlight, with an Earful-style XcodeGen and Fastlane setup, without changing the macOS app's behavior.

**Architecture:** Migrate the hand-built Xcode project to XcodeGen, then add an iOS app target and an iOS widget extension that compile the same `Shared/` code plus a new `WidgetShared/` folder (provider and views used by both widget extensions). On iOS the widget is the main fetcher, deciding with a new `RefreshPolicy.widgetShouldFetch`; the app refreshes when opened and from a best-effort background task. Fastlane gains a `platform :ios` block modeled on Earful; GitHub Actions gains an iOS simulator build and a manual TestFlight workflow.

**Tech Stack:** Swift 5 language mode, SwiftUI, WidgetKit, AppIntents, BackgroundTasks, Swift Testing, XcodeGen 2.46, Fastlane (match, produce, pilot), GitHub Actions on hosted `macos-15`.

**Spec:** `docs/superpowers/specs/2026-10-08-next-game-ios-design.md` (builds on `docs/superpowers/specs/2026-10-06-next-game-design.md`)

`Shared/AppGroup.swift`:

```swift
import Foundation
#if os(macOS)
import Security
#endif

/// The App Group shared by the app and the widget extension.
enum AppGroup {
    static let infoKey = "NextGameAppGroup"

    static let id: String? = {
        #if os(macOS)
        // Read from entitlements because the Team ID prefix is filled in at signing time.
        guard let task = SecTaskCreateFromSelf(nil),
              let value = SecTaskCopyValueForEntitlement(task, "com.apple.security.application-groups" as CFString, nil),
              let groups = value as? [String]
        else { return nil }
        return groups.first
        #else
        // iOS has no SecTask API, so the group comes from Info.plist, set per environment.
        return groupID(infoDictionary: Bundle.main.infoDictionary)
        #endif
    }()

    /// An unexpanded build setting means the xcconfig wasn't applied; treat it as no group rather than a bogus one.
    static func groupID(infoDictionary: [String: Any]?) -> String? {
        guard let value = infoDictionary?[infoKey] as? String, !value.isEmpty, !value.hasPrefix("$(") else { return nil }
        return value
    }

    /// Falls back to the process's own defaults when there's no App Group (unit tests).
    static let defaults: UserDefaults = id.flatMap { UserDefaults(suiteName: $0) } ?? .standard

    static let containerURL: URL = id.flatMap {
        FileManager.default.containerURL(forSecurityApplicationGroupIdentifier: $0)
    } ?? FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask)[0].appending(path: "NextGame")
}
```

import Foundation
import WidgetKit

enum WidgetTeams {
    static func configured() async -> [String] {
        let infos = (try? await WidgetCenter.shared.currentConfigurations()) ?? []
        let teams = infos.compactMap { $0.widgetConfigurationIntent(of: SelectTeamIntent.self)?.team?.id }
        return Array(Set(teams)).sorted()
    }
}

/// The refresh both apps run: teams weekly, logos, then the schedule.
struct AppRefresh {
    var refresher = ScheduleRefresher()
    var store = SnapshotStore.shared
    var logos = LogoCache.shared

    func run(configuredTeams: Set<String>) async -> RefreshOutcome {
        var logosChanged = false
        if let fresh = await refresher.refreshTeamsIfStale(maxAge: RefreshPolicy.teamsMaxAge) {
            logosChanged = await logos.update(fresh, force: true)
        } else if let teams = store.teams?.teams {
            logosChanged = await logos.update(teams)
        }

        let previous = store.schedule?.games
        let outcome = await refresher.refreshSchedule()
        // .changed fires for any game in the season; reloading on each would burn the widget budget.
        if logosChanged || (outcome == .changed
            && RefreshPolicy.relevantGamesChanged(old: previous, new: store.schedule?.games, teams: configuredTeams)) {
            WidgetCenter.shared.reloadAllTimelines()
        }
        return outcome
    }
}

    static let widgetMinimumInterval: TimeInterval = 15 * 60

    /// On macOS the menu bar app is always running and keeps the cache fresh; on iOS the app is usually
    /// suspended, so the widget is the main fetcher.
    #if os(macOS)
    static let appIsTheFetcher = true
    #else
    static let appIsTheFetcher = false
    #endif

    static func widgetShouldFetch(snapshotAge: TimeInterval?, team: String?, games: [Game], now: Date,
                                  appIsTheFetcher: Bool = appIsTheFetcher) -> Bool {
        guard let snapshotAge else { return true }
        if appIsTheFetcher { return snapshotAge >= widgetStaleAfter }
        let teams = Set([team].compactMap { $0 })
        return snapshotAge >= max(widgetMinimumInterval, interval(for: teams, games: games, now: now))

    static func inline(_ state: TeamState, team: String, now: Date,
                       calendar: Calendar = .current, locale: Locale = .current) -> String {
        switch state {
        case let .upcoming(next, _, _):
            return "\(team) \(matchup(next, for: team)) · \(kickoff(next.kickoff, now: now, calendar: calendar, locale: locale))"
        case let .inProgress(game):
            return game.hasScore ? liveScore(game, for: team) : "\(team) \(matchup(game, for: team)) · In progress"
        case .offseason:
            return "\(team) · Offseason"
        }
    }

    static func rectangularLines(_ state: TeamState, team: String, now: Date,
                                 calendar: Calendar = .current, locale: Locale = .current) -> [String] {
        switch state {
        case let .upcoming(next, last, isBye):

            let countdown = countdown(to: next.kickoff, now: now, calendar: calendar) + (isBye ? " · bye week" : "")
            return ["\(matchup(next, for: team)) · \(when)", countdown] + [last.map { result($0, for: team) }].compactMap { $0 }
        case let .inProgress(game):
            let header = "\(team) \(matchup(game, for: team))"
            if game.hasScore { return [header, liveScore(game, for: team)] }
            return [header, "In progress", kickoff(game.kickoff, now: now, calendar: calendar, locale: locale)]
        case let .offseason(last):
            return [team, "Offseason"] + [last.map { result($0, for: team) }].compactMap { $0 }

import BackgroundTasks
import Foundation
import Observation

@MainActor
@Observable
final class IOSAppModel {
    nonisolated static let refreshTaskID = (Bundle.main.bundleIdentifier ?? "biz.fitz.NextGame") + ".refresh"

    private(set) var schedule: ScheduleSnapshot?
    private(set) var configuredTeams: [String] = []
    private(set) var lastError: String?
    private(set) var isRefreshing = false
    private(set) var now = Date.now

    @ObservationIgnored private let store = SnapshotStore.shared

    init() {
        schedule = store.schedule
        lastError = schedule?.lastError
    }

    func state(for team: String) -> TeamState? {
        schedule.map { GameResolver.resolve(team: team, games: $0.games, now: now) }
    }

    var statusLine: String {
        GameFormatting.statusLine(fetchedAt: schedule?.fetchedAt, lastError: lastError, now: now)
    }

    /// The widget does most fetching on iOS; opening the app is a cheap extra (usually a 304).
    func becameActive() async {
        now = .now
        configuredTeams = await WidgetTeams.configured()
        await refresh()
    }

    func refresh() async {
        guard !isRefreshing else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        let outcome = await AppRefresh().run(configuredTeams: Set(configuredTeams))
        schedule = store.schedule
        now = .now
        if case let .failed(message) = outcome { lastError = message } else { lastError = nil }
    }

    /// iOS decides when, or whether, this runs; it only ever helps the widget's own fetching.
    func scheduleBackgroundRefresh() {
        let request = BGAppRefreshTaskRequest(identifier: Self.refreshTaskID)
        request.earliestBeginDate = .now + RefreshPolicy.interval(for: Set(configuredTeams), games: schedule?.games ?? [], now: .now)
        try? BGTaskScheduler.shared.submit(request)
    }

    func backgroundRefresh() async {
        configuredTeams = await WidgetTeams.configured()
        await refresh()
        scheduleBackgroundRefresh()
    }
}

```ruby
platform :ios do
  # Values live in the xcconfigs so the project and the lanes can't disagree.
  def xcconfig_value(env, key)
    path = File.expand_path("../Config/#{env == 'prod' ? 'Prod' : 'Dev'}.xcconfig", __dir__)
    line = File.readlines(path).find { |l| l.match?(/\A#{Regexp.escape(key)}\s*=/) }
    UI.user_error!("#{key} missing from #{path}") unless line
    line.split("=", 2).last.strip
  end

  def ios_ids(env)
    app = xcconfig_value(env, "NEXTGAME_IOS_BUNDLE_ID")
    [app, "#{app}.Widget"]
  end

  def apple_id
    ENV["FASTLANE_USER"].to_s.empty? ? UI.user_error!("Set FASTLANE_USER to your Apple ID") : ENV["FASTLANE_USER"]
  end

  desc "One-time, run locally: register the iOS App IDs and App Groups for prod and dev."
  lane :bootstrap do
    team = nextgame_team_id
    creds = "-u #{apple_id} -b #{team}"
    %w[prod dev].each do |env|
      group = xcconfig_value(env, "NEXTGAME_APP_GROUP")
      name = xcconfig_value(env, "NEXTGAME_DISPLAY_NAME")
      UI.header("#{env}: #{ios_ids(env).join(', ')}")
      sh("fastlane produce group #{creds} -g #{group} -n \"#{name} group\"")
      ios_ids(env).each do |bundle_id|
        produce(app_identifier: bundle_id, app_name: bundle_id.end_with?(".Widget") ? "#{name} Widget" : name,
                username: apple_id, team_id: team, skip_itc: true, sku: bundle_id, language: "en-US", platform: "ios")
        # Enable before associating: enabling App Groups clears the groups already selected on the App ID,
        # which yields a profile with an empty application-groups entitlement. Learned the hard way in Earful.
        sh("fastlane produce enable_services #{creds} -a #{bundle_id} --app-group")
        sh("fastlane produce associate_group #{creds} -a #{bundle_id} #{group}")
      end
    end
    UI.success("App IDs and groups ready. Next: bundle exec fastlane ios appstore_record, then ios certificates")
  end

  desc "Create the App Store Connect record TestFlight needs. Names are unique across the App Store;"
  desc "set NEXTGAME_ASC_NAME if 'Next Game' is taken. The home screen name comes from the xcconfig."
  lane :appstore_record do
    bundle_id = ios_ids("prod").first
    produce(app_identifier: bundle_id, app_name: ENV.fetch("NEXTGAME_ASC_NAME", "Next Game"),
            username: apple_id, team_id: nextgame_team_id, skip_devcenter: true,
            sku: bundle_id, language: "en-US", platform: "ios")
  end

  desc "Create or renew iOS development and App Store profiles in the match repo."
  lane :certificates do
    ids = ios_ids("prod") + ios_ids("dev")
    %w[development appstore].each do |type|
      match(type: type, platform: "ios", app_identifier: ids, readonly: false,
            skip_provisioning_profiles: false, api_key: asc_api_key)
    end
  end

  desc "Install iOS certificates and profiles on this Mac. Read-only."
  lane :setup do
    ids = ios_ids("prod") + ios_ids("dev")
    %w[development appstore].each do |type|
      match(type: type, platform: "ios", app_identifier: ids, readonly: true, skip_provisioning_profiles: false)
    end
  end

  desc "Build the prod app and upload it to TestFlight. Used by the TestFlight workflow."
  lane :beta do
    setup_ci if ENV["CI"]
    api_key = asc_api_key
    team = nextgame_team_id
    app_id, widget_id = ios_ids("prod")

    match(type: "appstore", platform: "ios", app_identifier: [app_id, widget_id], readonly: true,
          skip_provisioning_profiles: false, api_key: api_key, team_id: team)

    build_number = latest_testflight_build_number(api_key: api_key, app_identifier: app_id,
                                                  platform: "ios", initial_build_number: 0) + 1

    build_app(
      project: project_path,
      scheme: "NextGameiOS",
      configuration: "Release",
      export_method: "app-store",
      output_directory: build_dir,
      output_name: "NextGame",
      clean: true,
      xcargs: "DEVELOPMENT_TEAM=#{team} CURRENT_PROJECT_VERSION=#{build_number}",
      export_options: {
        method: "app-store",
        signingStyle: "manual",
        teamID: team,
        provisioningProfiles: {
          app_id => "match AppStore #{app_id}",
          widget_id => "match AppStore #{widget_id}"
        }
      }
    )

    upload_to_testflight(api_key: api_key, skip_waiting_for_build_processing: true, distribute_external: false)
  end
end
```
