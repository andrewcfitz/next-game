```swift
/// The App Group shared by the app and the widget extension.
enum AppGroup {
    /// Read from this process's entitlements because the Team ID prefix is filled in at signing time.
    static let id: String? = {
        guard let task = SecTaskCreateFromSelf(nil),
              let value = SecTaskCopyValueForEntitlement(task, "com.apple.security.application-groups" as CFString, nil),
              let groups = value as? [String]
        else { return nil }
        return groups.first
    }()

    /// Falls back to the process's own defaults when there's no App Group (unit tests).
    static let defaults: UserDefaults = id.flatMap { UserDefaults(suiteName: $0) } ?? .standard

    static let containerURL: URL = id.flatMap {
        FileManager.default.containerURL(forSecurityApplicationGroupIdentifier: $0)
    } ?? FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask)[0].appending(path: "NextGame")
}
```

```swift
import Foundation

struct Team: Codable, Hashable, Sendable, Identifiable {
    var abbr: String
    var name: String
    /// Hex, e.g. "#E31837".
    var primaryColor: String
    var secondaryColor: String
    var logoURL: URL?

    var id: String { abbr }
}

struct Game: Codable, Hashable, Sendable, Identifiable {
    var id: String
    var season: Int
    var week: Int
    /// nflverse codes: REG, WC, DIV, CON, SB.
    var gameType: String
    var kickoff: Date
    var awayTeam: String
    var homeTeam: String
    var awayScore: Int?
    var homeScore: Int?
    var stadium: String?
    var spread: Double?

    /// nflverse has no end time, so assume a typical broadcast window.
    static let expectedDuration: TimeInterval = 3.5 * 3600
    /// No score this long after kickoff means postponed or bad data; stop treating it as live.
    static let staleAfter: TimeInterval = 24 * 3600

    var expectedEnd: Date { kickoff + Self.expectedDuration }
    var hasScore: Bool { awayScore != nil && homeScore != nil }

    func involves(_ team: String) -> Bool { awayTeam == team || homeTeam == team }
    func isHome(_ team: String) -> Bool { homeTeam == team }
    func opponent(of team: String) -> String { homeTeam == team ? awayTeam : homeTeam }
    func score(of team: String) -> Int? { homeTeam == team ? homeScore : awayScore }

    /// Scores can appear before the game ends if nflverse posts live scores, so also wait out the window.
    func isFinal(at now: Date) -> Bool { hasScore && now >= expectedEnd }
}

enum TeamState: Equatable, Sendable {
    case upcoming(next: Game, last: Game?, isBye: Bool)
    case inProgress(Game)
    case offseason(last: Game?)
}
```

```swift
import Foundation

enum ParseError: Error, Equatable {
    case notUTF8
    case missingColumn(String)
    case noRows
}

enum CSV {
    /// RFC 4180: quoted fields, doubled quotes, commas and newlines inside quotes, LF or CRLF.
    static func parse(_ text: String) -> [[String]] {
        var rows: [[String]] = []
        var row: [String] = []
        var field = String.UnicodeScalarView()
        var inQuotes = false
        let scalars = Array(text.unicodeScalars)
        var i = 0
        while i < scalars.count {
            let c = scalars[i]
            if inQuotes {
                if c == "\"" {
                    if i + 1 < scalars.count, scalars[i + 1] == "\"" {
                        field.append("\"")
                        i += 1
                    } else {
                        inQuotes = false
                    }
                } else {
                    field.append(c)
                }
            } else {
                switch c {
                case "\"": inQuotes = true
                case ",":
                    row.append(String(field))
                    field = String.UnicodeScalarView()
                case "\n":
                    row.append(String(field))
                    rows.append(row)
                    row = []
                    field = String.UnicodeScalarView()
                case "\r": break
                default: field.append(c)
                }
            }
            i += 1
        }
        if !field.isEmpty || !row.isEmpty {
            row.append(String(field))
            rows.append(row)
        }
        return rows
    }

    /// Maps header names to column indexes.
    static func columns(_ header: [String]) -> [String: Int] {
        Dictionary(header.enumerated().map { ($1, $0) }, uniquingKeysWith: { first, _ in first })
    }
}
```

```swift
import Foundation

enum ScheduleParser {
    static let eastern = TimeZone(identifier: "America/New_York")!
    /// Older seasons and some flexed games have no time yet; Sunday early slot is the best guess.
    static let defaultKickoff = "13:00"

    /// Parses nflverse games.csv, keeping only the latest season in the file.
    static func parse(_ data: Data) throws -> (season: Int, games: [Game]) {
        guard let text = String(data: data, encoding: .utf8) else { throw ParseError.notUTF8 }
        // The file covers every season since 1999; filter lines by the game_id prefix before
        // the full CSV parse to keep memory low in the widget extension.
        let lines = text.split(separator: "\n")
        guard let header = lines.first else { throw ParseError.noRows }
        guard let season = lines.dropFirst().compactMap({ Int($0.prefix(4)) }).max() else { throw ParseError.noRows }
        let prefix = "\(season)_"
        let body = lines.dropFirst().filter { $0.hasPrefix(prefix) }
        let rows = CSV.parse(([header] + body).joined(separator: "\n"))

        let columns = CSV.columns(rows[0])
        func index(_ name: String) throws -> Int {
            guard let i = columns[name] else { throw ParseError.missingColumn(name) }
            return i
        }
        let id = try index("game_id"), seasonCol = try index("season"), type = try index("game_type")
        let week = try index("week"), day = try index("gameday"), time = try index("gametime")
        let away = try index("away_team"), awayScore = try index("away_score")
        let home = try index("home_team"), homeScore = try index("home_score")
        let stadium = try index("stadium"), spread = try index("spread_line")

        var games: [Game] = []
        for row in rows.dropFirst() where row.count == rows[0].count {
            guard let rowSeason = Int(row[seasonCol]), let rowWeek = Int(row[week]),
                  let kickoff = kickoffDate(day: row[day], time: row[time])
            else { continue }
            games.append(Game(
                id: row[id], season: rowSeason, week: rowWeek, gameType: row[type], kickoff: kickoff,
                awayTeam: row[away], homeTeam: row[home],
                awayScore: Int(row[awayScore]), homeScore: Int(row[homeScore]),

                spread: Double(row[spread])
            ))
        }
        guard !games.isEmpty else { throw ParseError.noRows }
        return (season, games.sorted { $0.kickoff < $1.kickoff })

    static func kickoffDate(day: String, time: String) -> Date? {

        calendar.timeZone = eastern
        return calendar.date(from: DateComponents(year: d[0], month: d[1], day: d[2], hour: t[0], minute: t[1]))
```

```swift
import Foundation

/// Decides what a team's widget shows at a given moment. Pure, so timelines can ask about future dates.
enum GameResolver {
    static func resolve(team: String, games: [Game], now: Date) -> TeamState {
        let mine = teamGames(team, games)
        if let live = mine.first(where: { isLive($0, at: now) }) {
            return .inProgress(live)
        }
        let last = mine.last { $0.isFinal(at: now) }
        guard let next = mine.first(where: { $0.kickoff > now }) else {
            return .offseason(last: last)
        }
        // Never pair last season's final game with this season's opener.
        let shownLast = last?.season == next.season ? last : nil
        return .upcoming(next: next, last: shownLast, isBye: isBye(team: team, last: shownLast, next: next, games: games, now: now))
    }

    static func isLive(_ game: Game, at now: Date) -> Bool {
        game.kickoff <= now && !game.isFinal(at: now) && now < game.kickoff + Game.staleAfter
    }

    static func laterGames(team: String, games: [Game], now: Date, limit: Int = 2) -> [Game] {
        let mine = teamGames(team, games)
        guard let featured = mine.first(where: { isLive($0, at: now) || $0.kickoff > now }) else { return [] }
        return Array(mine.filter { $0.kickoff > featured.kickoff }.prefix(limit))
    }

    /// Moments the widget's state can change without new data: kickoffs, expected ends, and week starts (bye tag).
    static func timelineDates(team: String, games: [Game], now: Date) -> [Date] {
        var dates: Set<Date> = []
        for game in teamGames(team, games) {
            for date in [game.kickoff, game.expectedEnd] where date > now {
                dates.insert(date)
            }
        }
        for start in weekStarts(games).values where start > now {
            dates.insert(start)
        }
        return [now] + dates.sorted().prefix(7)
    }

    private static func teamGames(_ team: String, _ games: [Game]) -> [Game] {
        games.filter { $0.involves(team) }.sorted { $0.kickoff < $1.kickoff }
    }

    private struct WeekKey: Hashable {
        var season: Int
        var week: Int
    }

    private static func weekStarts(_ games: [Game]) -> [WeekKey: Date] {
        var starts: [WeekKey: Date] = [:]
        for game in games {
            let key = WeekKey(season: game.season, week: game.week)
            starts[key] = min(starts[key] ?? game.kickoff, game.kickoff)
        }
        return starts
    }

    private static func isBye(team: String, last: Game?, next: Game, games: [Game], now: Date) -> Bool {
        guard let last, next.week > last.week + 1 else { return false }
        let byeWeek = last.week + 1
        if games.contains(where: { $0.involves(team) && $0.season == next.season && $0.week == byeWeek }) { return false }
        let starts = weekStarts(games)
        guard let byeStart = starts[WeekKey(season: next.season, week: byeWeek)] else { return false }
        let nextWeekStart = starts[WeekKey(season: next.season, week: next.week)] ?? next.kickoff
        return now >= byeStart && now < nextWeekStart
    }
}
```

```swift
import Foundation

enum GameFormatting {
    static func matchup(_ game: Game, for team: String) -> String {
        "\(game.isHome(team) ? "vs" : "@") \(game.opponent(of: team))"
    }

    static func result(_ game: Game, for team: String) -> String {
        guard let mine = game.score(of: team), let theirs = game.score(of: game.opponent(of: team)) else {
            return matchup(game, for: team)
        }
        let letter = mine > theirs ? "W" : mine < theirs ? "L" : "T"
        return "\(letter) \(mine)-\(theirs) \(matchup(game, for: team))"
    }

    static func liveScore(_ game: Game, for team: String) -> String {
        let opponent = game.opponent(of: team)
        guard let mine = game.score(of: team), let theirs = game.score(of: opponent) else { return "In progress" }
        return "\(team) \(mine), \(opponent) \(theirs)"
    }

    static func kickoff(_ date: Date, now: Date, calendar: Calendar = .current, locale: Locale = .current) -> String {
        let style = Date.FormatStyle(locale: locale, calendar: calendar, timeZone: calendar.timeZone)
        let time = date.formatted(style.hour().minute())
        let days = dayDistance(from: now, to: date, calendar: calendar)
        let day: String
        switch days {
        case 0: day = "Today"
        case 1: day = "Tomorrow"
        case 2..<7: day = date.formatted(style.weekday(.abbreviated))
        default: day = date.formatted(style.month(.abbreviated).day())
        }
        return "\(day) \(time)"
    }

    static func countdown(to date: Date, now: Date, calendar: Calendar = .current) -> String {
        let seconds = date.timeIntervalSince(now)
        if seconds <= 0 { return "now" }
        if seconds < 3600 { return "in \(max(1, Int(seconds / 60)))m" }
        let days = dayDistance(from: now, to: date, calendar: calendar)
        if days == 0 { return "in \(Int(seconds / 3600))h" }
        return "in \(days)d"
    }

    private static func dayDistance(from now: Date, to date: Date, calendar: Calendar) -> Int {
        calendar.dateComponents([.day], from: calendar.startOfDay(for: now), to: calendar.startOfDay(for: date)).day ?? 0
    }
}
```

```swift
    var requests: [URLRequest] { lock.withLock { recorded } }

        let reply: Reply = lock.withLock {
            recorded.append(request)
            return replies.isEmpty ? .failure(.resourceUnavailable) : replies.removeFirst()
        }
        switch reply {
        case let .status(code, data, headers):
            return (data, HTTPURLResponse(url: request.url!, statusCode: code, httpVersion: nil, headerFields: headers)!)
        case let .failure(code):
            throw URLError(code)
        }
```

```swift
enum FetchResult: Equatable, Sendable {
    case notModified
    case updated(Data, etag: String?)
}

struct HTTPError: Error, Equatable {
    var status: Int
}

struct NFLVerseClient: Sendable {
    static let scheduleURL = URL(string: "https://raw.githubusercontent.com/nflverse/nfldata/master/data/games.csv")!
    static let teamsURL = URL(string: "https://github.com/nflverse/nflverse-data/releases/download/teams/teams_colors_logos.csv")!

    var transport: @Sendable (URLRequest) async throws -> (Data, URLResponse) = {
        try await URLSession.shared.data(for: $0)
    }

    func fetch(_ url: URL, etag: String? = nil) async throws -> FetchResult {
        // Skip URLSession's own cache so a 304 reaches us instead of being turned back into a 200.
        var request = URLRequest(url: url, cachePolicy: .reloadIgnoringLocalCacheData, timeoutInterval: 30)
        if let etag { request.setValue(etag, forHTTPHeaderField: "If-None-Match") }
        let (data, response) = try await transport(request)
        let http = response as? HTTPURLResponse
        switch http?.statusCode ?? 0 {
        case 304: return .notModified
        case 200..<300: return .updated(data, etag: http?.value(forHTTPHeaderField: "ETag"))
        case let status: throw HTTPError(status: status)
        }
    }
}
```

```swift
struct ScheduleSnapshot: Codable, Equatable, Sendable {
    var season: Int
    var games: [Game]
    var etag: String?
    var fetchedAt: Date
    var lastError: String?
}

struct TeamsSnapshot: Codable, Equatable, Sendable {
    var teams: [Team]
    var fetchedAt: Date
}

/// Cached data shared by the app and widget through App Group defaults.
struct SnapshotStore: @unchecked Sendable {
    static let shared = SnapshotStore(defaults: AppGroup.defaults)

    let defaults: UserDefaults

    init(defaults: UserDefaults) {
        self.defaults = defaults
    }

    var schedule: ScheduleSnapshot? {
        get { load(ScheduleSnapshot.self, key: "schedule") }
        nonmutating set { save(newValue, key: "schedule") }
    }

    var teams: TeamsSnapshot? {
        get { load(TeamsSnapshot.self, key: "teams") }
        nonmutating set { save(newValue, key: "teams") }
    }

    private func load<T: Decodable>(_ type: T.Type, key: String) -> T? {
        guard let data = defaults.data(forKey: key) else { return nil }
        return try? JSONDecoder().decode(type, from: data)
    }

    private func save<T: Encodable>(_ value: T?, key: String) {
        guard let value, let data = try? JSONEncoder().encode(value) else {
            defaults.removeObject(forKey: key)
            return
        }
        defaults.set(data, forKey: key)
    }
}
```

```swift
import Foundation

enum RefreshOutcome: Equatable, Sendable {
    case unchanged
    case changed
    case failed(String)
}

/// Downloads nflverse data into the snapshot store. Never replaces good data with bad.
struct ScheduleRefresher: Sendable {
    var client = NFLVerseClient()
    var store = SnapshotStore.shared
    var now: @Sendable () -> Date = { .now }

    func refreshSchedule() async -> RefreshOutcome {
        let previous = store.schedule
        do {
            switch try await client.fetch(NFLVerseClient.scheduleURL, etag: previous?.etag) {
            case .notModified:
                guard var snapshot = previous else { return .unchanged }
                snapshot.fetchedAt = now()
                snapshot.lastError = nil
                store.schedule = snapshot
                return .unchanged
            case let .updated(data, etag):
                let parsed = try ScheduleParser.parse(data)
                store.schedule = ScheduleSnapshot(season: parsed.season, games: parsed.games, etag: etag,
                                                  fetchedAt: now(), lastError: nil)
                return parsed.games == previous?.games ? .unchanged : .changed
            }
        } catch {
            let message = Self.describe(error)
            if var snapshot = previous {
                snapshot.lastError = message
                store.schedule = snapshot
            }
            return .failed(message)
        }
    }

    func refreshTeamsIfStale(maxAge: TimeInterval = 7 * 86400) async -> [Team]? {
        if let cached = store.teams, now().timeIntervalSince(cached.fetchedAt) < maxAge { return nil }
        guard case let .updated(data, _) = try? await client.fetch(NFLVerseClient.teamsURL),
              let teams = try? TeamsParser.parse(data)
        else { return nil }
        store.teams = TeamsSnapshot(teams: teams, fetchedAt: now())
        return teams
    }

    static func describe(_ error: Error) -> String {
        switch error {
        case let error as HTTPError: "HTTP \(error.status)"
        case is ParseError: "Couldn't read the schedule"
        default: error.localizedDescription
        }
    }
}
```

```swift
import AppKit
```

```swift
/// Team logos on disk in the App Group, because widgets can't load remote images while rendering.
struct LogoCache: Sendable {
    static let shared = LogoCache(directory: AppGroup.containerURL.appending(path: "Logos", directoryHint: .isDirectory))

    let directory: URL
    var download: @Sendable (URL) async throws -> Data

    init(directory: URL, download: @escaping @Sendable (URL) async throws -> Data = { try await URLSession.shared.data(from: $0).0 }) {
        self.directory = directory
        self.download = download
    }

    func fileURL(for abbr: String) -> URL {
        directory.appending(path: "\(abbr).png")
    }

    func image(for abbr: String) -> NSImage? {
        NSImage(contentsOf: fileURL(for: abbr))
    }

    @discardableResult
    func update(_ teams: [Team], force: Bool = false) async -> Bool {
        try? FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        return await withTaskGroup(of: Bool.self) { group in
            for team in teams {
                guard let url = team.logoURL else { continue }
                let file = fileURL(for: team.abbr)
                if !force, FileManager.default.fileExists(atPath: file.path()) { continue }
                group.addTask { [download] in
                    guard let data = try? await download(url), let png = Self.resized(data) else { return false }
                    return (try? png.write(to: file, options: .atomic)) != nil
                }
            }
            return await group.reduce(false) { $0 || $1 }
        }
    }

    /// ESPN serves 500px logos; widget extensions have tight memory limits.
    static func resized(_ data: Data, side: Int = 128) -> Data? {
        guard let source = CGImageSourceCreateWithData(data as CFData, nil),
              let image = CGImageSourceCreateThumbnailAtIndex(source, 0, [
                  kCGImageSourceCreateThumbnailFromImageAlways: true,
                  kCGImageSourceThumbnailMaxPixelSize: side,
                  kCGImageSourceCreateThumbnailWithTransform: true,
              ] as CFDictionary)
        else { return nil }
        let output = NSMutableData()
        guard let destination = CGImageDestinationCreateWithData(output, UTType.png.identifier as CFString, 1, nil) else { return nil }
        CGImageDestinationAddImage(destination, image, nil)
        guard CGImageDestinationFinalize(destination) else { return nil }
        return output as Data
    }
}
```

```swift
import Foundation

/// nflverse commits every 10 to 20 minutes, so polling faster than that only costs bandwidth.
enum RefreshPolicy {
    static let normal: TimeInterval = 6 * 3600
    static let gameDay: TimeInterval = 10 * 60
    static let leadTime: TimeInterval = 4 * 3600
    static let widgetStaleAfter: TimeInterval = 12 * 3600
    static let teamsMaxAge: TimeInterval = 7 * 86400

    static func interval(for teams: Set<String>, games: [Game], now: Date) -> TimeInterval {
        let relevant = games.filter { teams.contains($0.awayTeam) || teams.contains($0.homeTeam) }
        if relevant.contains(where: { isActive($0, at: now) }) { return gameDay }
        guard let window = relevant.map({ $0.kickoff - leadTime }).filter({ $0 > now }).min() else { return normal }
        return max(gameDay, min(normal, window.timeIntervalSince(now)))
    }

    static func isDue(lastAttempt: Date?, teams: Set<String>, games: [Game], now: Date) -> Bool {
        guard let lastAttempt else { return true }
        return now >= lastAttempt + interval(for: teams, games: games, now: now)
    }

    private static func isActive(_ game: Game, at now: Date) -> Bool {
        now >= game.kickoff - leadTime && !game.isFinal(at: now) && now < game.kickoff + Game.staleAfter
    }
}
```

```swift
struct Provider: AppIntentTimelineProvider {
    func placeholder(in context: Context) -> Entry { .sample }

    func snapshot(for configuration: SelectTeamIntent, in context: Context) async -> Entry {
        let schedule = SnapshotStore.shared.schedule
        if context.isPreview, schedule == nil || configuration.team == nil { return .sample }
        return entry(team: configuration.team?.id, schedule: schedule, at: .now)
    }

    func timeline(for configuration: SelectTeamIntent, in context: Context) async -> Timeline<Entry> {
        let schedule = await loadSchedule()
        let now = Date.now
        let team = configuration.team?.id
        let games = schedule?.games ?? []
        let dates = team.map { GameResolver.timelineDates(team: $0, games: games, now: now) } ?? [now]
        let wait = RefreshPolicy.interval(for: Set([team].compactMap { $0 }), games: games, now: now)
        // WidgetKit throttles reloads below about 15 minutes anyway.
        return Timeline(entries: dates.map { entry(team: team, schedule: schedule, at: $0) },
                        policy: .after(now + max(wait, 15 * 60)))
    }

    /// The app normally keeps the snapshot fresh; fetch here only if it hasn't run in a while.
    private func loadSchedule() async -> ScheduleSnapshot? {
        let store = SnapshotStore.shared
        if let cached = store.schedule, Date.now.timeIntervalSince(cached.fetchedAt) < RefreshPolicy.widgetStaleAfter {
            return cached
        }
        _ = await ScheduleRefresher().refreshSchedule()
        return store.schedule
    }

    private func entry(team: String?, schedule: ScheduleSnapshot?, at date: Date) -> Entry {
        let teams = Dictionary((SnapshotStore.shared.teams?.teams ?? []).map { ($0.abbr, $0) }, uniquingKeysWith: { a, _ in a })

            return Entry(date: date, team: team, state: nil, later: [], teams: teams, hasSchedule: schedule != nil)

        return Entry(date: date, team: team,
                     state: GameResolver.resolve(team: team, games: schedule.games, now: date),
                     later: GameResolver.laterGames(team: team, games: schedule.games, now: date),
                     teams: teams, hasSchedule: true)
    }
```

```swift
import AppKit
import Observation
import ServiceManagement
import WidgetKit

@MainActor
@Observable
final class AppModel {
    private(set) var schedule: ScheduleSnapshot?
    private(set) var configuredTeams: [String] = []
    private(set) var lastError: String?
    private(set) var isRefreshing = false
    private(set) var launchAtLogin = SMAppService.mainApp.status == .enabled
    private(set) var now = Date.now

    @ObservationIgnored private let store = SnapshotStore.shared
    @ObservationIgnored private let refresher = ScheduleRefresher()
    @ObservationIgnored private var lastAttempt: Date?
    @ObservationIgnored private var loop: Task<Void, Never>?
    @ObservationIgnored private var observers: [NSObjectProtocol] = []

    init() {
        schedule = store.schedule
        lastError = schedule?.lastError
        startLoop()
        observeWake()
    }

    func state(for team: String) -> TeamState? {
        schedule.map { GameResolver.resolve(team: team, games: $0.games, now: now) }
    }

    var statusLine: String {
        if let lastError { return lastError }
        guard let fetchedAt = schedule?.fetchedAt else { return "Not updated yet" }
        return "Updated \(fetchedAt.formatted(.relative(presentation: .named)))"
    }

    func menuOpened() async {
        now = .now
        configuredTeams = await Self.loadConfiguredTeams()
    }

    func refresh() async {
        guard !isRefreshing else { return }
        isRefreshing = true
        defer { isRefreshing = false }
        lastAttempt = .now

        var logosChanged = false
        if let fresh = await refresher.refreshTeamsIfStale(maxAge: RefreshPolicy.teamsMaxAge) {
            logosChanged = await LogoCache.shared.update(fresh, force: true)
        } else if let teams = store.teams?.teams {
            logosChanged = await LogoCache.shared.update(teams)
        }

        let outcome = await refresher.refreshSchedule()
        schedule = store.schedule
        now = .now
        switch outcome {
        case .changed, .unchanged: lastError = nil
        case let .failed(message): lastError = message
        }
        if outcome == .changed || logosChanged {
            WidgetCenter.shared.reloadAllTimelines()
        }
    }

    /// Ticks often but only fetches when due, so a widget added minutes before kickoff gets game-day refresh.
    private func startLoop() {
        loop = Task { [weak self] in
            while !Task.isCancelled {
                guard let self else { return }
                configuredTeams = await Self.loadConfiguredTeams()
                if RefreshPolicy.isDue(lastAttempt: lastAttempt, teams: Set(configuredTeams),
                                       games: schedule?.games ?? [], now: .now) {
                    await refresh()
                }
                try? await Task.sleep(for: .seconds(5 * 60))
            }
        }
    }

    private func observeWake() {
        observers.append(NSWorkspace.shared.notificationCenter.addObserver(
            forName: NSWorkspace.didWakeNotification, object: nil, queue: .main
        ) { [weak self] _ in
            Task { @MainActor in await self?.refresh() }
        })
    }

    private static func loadConfiguredTeams() async -> [String] {
        let infos = (try? await WidgetCenter.shared.currentConfigurations()) ?? []
        let teams = infos.compactMap { $0.widgetConfigurationIntent(of: SelectTeamIntent.self)?.team?.id }
        return Array(Set(teams)).sorted()
    }

    func setLaunchAtLogin(_ enabled: Bool) {
        do {
            if enabled {
                try SMAppService.mainApp.register()
            } else {
                try SMAppService.mainApp.unregister()
            }
        } catch {
            NSLog("Next Game: couldn't update login item: \(error)")
        }
        launchAtLogin = SMAppService.mainApp.status == .enabled
    }
}
```
